% ------ PROBLEM 2 ------ %
\begin{homeworkProblem}

  Suppose that $X$ has a density of exponential form
  \[
    f_\theta(x) \propto_{x} h(x) \exp \bp{\sum_{i=1}^{d} \theta_i T_i(x) - A(\theta)}
  \]
  where each $T_i(x)$ is differentiable.

  \begin{enumerate}[label=(\roman*), topsep=5pt, itemsep=5pt]
    \item Show that if $X$ has support in real line and $g$ is any differentiable function such that $\E_\theta[g^{\prime}(X)] < \infty$, then
          \[
            \E_\theta \left[ \bp{\frac{h^{\prime}(X)}{h(X)} + \sum_{i=1}^{d} \theta_i T_i^{\prime}(X)} g(X) \right] = - \E_\theta[g^{\prime}(X)].
          \]
    \item Using part a), show that if $X \sim \Nc(\mu, \sigma^2)$, then
          \[
            \Cov_\theta[g(X), X] = \sigma^2 \E_\theta[g^{\prime} (X)]
          \]
          where $\theta = [\mu, \sigma^2]^t$.
    \item Use part ii) to compute the third and fourth moments of $\Nc(\mu, \sigma^2)$. \newline
  \end{enumerate}

  \solution

\end{homeworkProblem}

\begin{homeworkProblem}

  Let
  \[
    f_\theta(x) = h(x) \exp(\theta x - A(\theta))
  \]
  be a one-parameter, canonical exponential family.

  \begin{enumerate}[label=(\roman*), topsep=5pt, itemsep=5pt]
    \item Show that for any prior $\pi(\theta)$,
          \[
            \E_\pi [\Theta \mid X = x] = \frac{d}{dx} \log \bp{\frac{m_\pi(x)}{h(x)}},
          \]
          where $m_\pi(x)$ is the marginal distribution of $X$.
    \item Use part i) to prove Tweedie's formula:
          \[
            \E_\pi [ \Theta \mid X = x] = x + \frac{d}{dx} \log m_\pi(x),
          \]
          when $f_\theta$ is $\Nc(\theta, 1)$. \newline
  \end{enumerate}

\end{homeworkProblem}
