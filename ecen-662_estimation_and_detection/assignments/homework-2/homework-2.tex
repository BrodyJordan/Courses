% -- PROBLEM 1 -- %
\begin{homeworkProblem}

  \begin{enumerate}[label=\roman*), topsep=5pt, itemsep=5pt]
    \item Let
          \[
            f_\theta(x) = h(x) \exp (\eta^t T - A(\eta))
          \]
          be an $s$-parameter exponential family of distributions. Show that if there exists
          \[
            \{\theta_0, \theta_1, \dots, \theta_s \} \subseteq \Theta
          \]
          such that the vectors
          \[
            (\eta(\theta_i) - \eta(\theta_0) : i \in [s])
          \]
          are linearly independent in $\mathcal{R}^s$, then $T$ is a minimal sufficient statistic.
    \item Let $X = (X_1, \dots, X_n)$, where $X_i$'s are i.i.d. $\mathcal{N}(\theta, \theta^2)$, and $\theta>0$ is the unknown parameter. Use the result of part i) to show that
          \[
            T = \left( \sum_{i=1}^n X_i, \sum_{i=1}^n X_i^2 \right)
          \]
          is a minimal sufficient statistic. \\
  \end{enumerate}

  \solution

  \textbf{Part (i):} Since the ratio $f_\theta(x) / f_{\theta}(y)$ is independent of $\theta$, and it implies that $T(x) = T(y)$, $T$ is a minimal sufficient statistic. \\

  \textbf{Part (ii):} Here we have pdf
  \[
    f_\theta(x) = \frac{1}{\sqrt{2\pi\theta^2}} \exp \left( -\frac{(x-\theta)^2}{2\theta^2} \right) = \frac{1}{\sqrt{2\pi\theta^2}} \exp \left( -\frac{x^2}{2\theta^2} + \frac{x}{\theta} - \frac{1}{2} \right)
  \]
  so $T = (X, X^2)$ is a sufficient statistic.


\end{homeworkProblem}

% -- PROBLEM 2 -- %
\begin{homeworkProblem}

  Suppose that $T_1$ is sufficient and $T_2$ is minimally sufficient. Let $U$ be an unbiased
  estimator of $\theta$ and define
  \[
    U_1 = \E_\theta[U|T_1] \quad \text{and} \quad U_2 = \E_\theta[U|T_2]
  \]
  Show that $U_2 = \E_\theta[U_1|T_2]$ and hence $\Var_\theta[U_2] \leq \Var_\theta[U_1]$. \\

  \solution

  Using the definition of $U_2$
  \[
    U_2 = \E_\theta[U|T_2] = \E_\theta[\E_\theta[U|T_1]|T_2] = \E_\theta[U_1|T_2]
  \]
  and for the variance
  \[
    \Var_\theta[U_1] = \Var(\E_\theta[U_1|T_2]) + \E_\theta[\Var_\theta[U_1|T_2]] = \Var[U_2] + \E_\theta[\Var_\theta[U_1|T_2]] \geq \Var_\theta[U_2]
  \]


\end{homeworkProblem}

\begin{homeworkProblem}

  Let $X = (X_1, \dots, X_n)$, where $X_i$'s are i.i.d. $\mathcal{U}([0, \theta])$, and $\theta > 0$ is
  the unknown parameter. Suppose $g$ is a differentiable function on $\mathcal{R}_{++}$. Find an UMVUE of $g(\theta)$
  in terms of the function $g$. \\

  \solution

  Here the pdf is
  \[
    f_\theta(t) = nt^{n-1} \theta^{-n} 1_{\{ 0 \leq t \leq \theta\} }
  \]
  and the expectation is
  \[
    \E_\theta[T] = \int_0^\theta nt^n \theta^{-n} dt = \frac{n\theta}{n+1}
  \]
  so then $\hat{\theta} = \frac{n+1}{n} T$ must be an UMVUE of $\theta$.
\end{homeworkProblem}

% -- PROBLEM 4 -- %
\begin{homeworkProblem}

  Let $X = [X_1, \dots, X_n]^t$, where
  \[
    X_i \iid Poisson(\lambda),
  \]
  and $\lambda > 0$ is the unknown parameter. Suppose we wish to estimate
  \[
    \theta = \exp(-\lambda) = \mathbb{P}_\lambda(X_i = 0).
  \]
  \begin{enumerate}[label=\roman*), topsep=5pt, itemsep=5pt]
    \item Show that $T = \sum_{i=1}^n X_i$ is a complete sufficient statistic for this likelihood family.
    \item Show that $S_1 = 1_{\{ X_1 = 0 \} }$ and $S_2 = \frac{1}{n} \sum_{i=1}^n 1_{\{ X_i = 0 \} }$ are both unbiased estimate of $\theta$.
    \item Compute the Rao-Blackwellized estimators $\tilde{S}_i = \E_\theta[S_i | T]$, $i = 1, 2$. What does your answer have to do with the completeness of $T$? \newline
  \end{enumerate}

  \solution

  \textbf{Part (i):} Here the pdf is
  \[
    f_\lambda(x) = \prod_{i=1}^n \frac{\lambda^{x_i} e^{-\lambda}}{x_i!} = \frac{\lambda^{\sum_{i=1}^n x_i} e^{-n\lambda}}{\prod_{i=1}^n x_i!}
  \]
  so $T = \sum_{i=1}^n X_i$ is a sufficient statistic. \\

  \textbf{Part (ii):} The expected value of $S_1$ is
  \[
    \E_\lambda[S_1] = \mathbb{P}_\lambda(X_1 = 0) = \exp(-\lambda) = \theta
  \]
  and the expected value of $S_2$ is
  \[
    \E_\lambda[S_2] = \frac{1}{n} \sum_{i=1}^n \mathbb{P}_\lambda(X_i = 0) = \exp(-\lambda) = \theta
  \]
  so both $S_1$ and $S_2$ are unbiased estimators of $\theta$. \\

  \textbf{Part (iii):} Here
  \[
    \tilde{S}_1 = \E_\theta[S_1 | T] = \E_\theta[S_2 | T] = \tilde{S}_2
  \]
  thus the estimators are equal.

\end{homeworkProblem}
