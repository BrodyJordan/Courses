% ----- Problem 2 ------ %
\begin{homeworkProblem}

Let $X = $ "I am happy," let $Y = $ "I am watching a movie" and $Z = $ "I am eating spaghetti." 
Translate the following statements into words.

\begin{enumerate}[label=(\alph*), topsep=5pt, itemsep=2pt]
    \item $Z \rightarrow X$.
    \item $X \leftrightarrow Y$.
    \item $(Y \lor Z) \rightarrow X$.
    \item $Y \lor (Z \rightarrow X)$.
    \item $(Y \rightarrow \lnot X) \land (Z \rightarrow \lnot X)$.
    \item $(X \land \lnot Y) \leftrightarrow (Y \lor Z)$.
\end{enumerate} \vspace{1\baselineskip}

\solution

\textbf{Part (a):} If I am eating spaghetti, then I am happy. \\

\textbf{Part (b):} I am happy if and only if I am watching a movie. \\

\textbf{Part (c):} If I am watching a movie or I am eating spaghetti, then I am happy. \\
 
\textbf{Part (d):} If I am eating spaghetti then I am happy, or I am watching a movie. \\

\textbf{Part (e):} If I am watching a movie then I am not happy, and if I am eating spaghetti then I am not happy. \\

\textbf{Part (f):} I am happy and I am not watching a movie, if and only if I am watching a movie or I am eating spaghetti. 

\end{homeworkProblem}

% ----- Problem 3 ------ %
\begin{homeworkProblem}

Let $E = $ "The house is blue," let $F = $ "The house is 30 years old" and $G = $ "The house is ugly." Translate the following statements into symbols.

\begin{enumerate}[label=(\alph*), topsep=5pt, itemsep=2pt]
    \item If the house is 30 years old, then it is ugly.
    \item If the house is blue, then it is ugly or it is 30 years old.
    \item If the house is blue, then it is ugly, or it is 30 years old.
    \item The house is not ugly if and only if it is 30 years old.
    \item The house is 30 years old if it is blue, and it is not ugly if it is 30 years old.
    \item For the house to be ugly, it is necessary and sufficient that it be ugly and 30 years old.
\end{enumerate} \vspace{1\baselineskip}

\solution

\textbf{Part (a):} $F \rightarrow G$.\\

\textbf{Part (b):} $E \rightarrow (G \lor F)$ \\

\textbf{Part (c):} $(E \rightarrow G) \lor F$ \\
 
\textbf{Part (d):} $\lnot G \leftrightarrow F$ \\

\textbf{Part (e):} $(E \rightarrow F) \land (F \rightarrow \lnot G)$ \\

\textbf{Part (f):} $G \leftrightarrow (G \land F)$

\end{homeworkProblem}

% ----- Problem 5 ------ %
\begin{homeworkProblem}

Simplify the following statements (making use of any equivalance of statements given so far in the text or exercises).

\begin{enumerate}[label=(\alph*), topsep=5pt, itemsep=2pt]
    \item $\lnot (P \rightarrow \lnot Q)$
    \item $A \rightarrow (A \land B)$
    \item $(X \land Y) \rightarrow X$
    \item $\lnot(M \lor L) \land L$
    \item $(P \rightarrow Q) \lor Q$
    \item $\lnot (X \rightarrow Y) \lor Y$
\end{enumerate} \vspace{1\baselineskip}

\solution

These simplifications can be found pretty quickly by writing out the truth tables, though I will omit that process and use the stated identities in the text and notes. \vspace{1\baselineskip}

\textbf{Part (a):} $\lnot (P \rightarrow \lnot Q) \equiv P \land \lnot (\lnot Q) \equiv P \land Q$. \\

\textbf{Part (b):} $A \rightarrow (A \land B) \equiv \lnot A \lor (A \land B) \equiv (\lnot A \lor A) \land (\lnot A \lor B) \equiv \lnot A \lor B$. \\

\textbf{Part (c):} $(X \land Y) \rightarrow X \equiv \lnot(X \land Y) \lor X \equiv \mathord{\underbrace{(\lnot X \lor X)}_{\text{tautology}}} \lor (\lnot Y) \equiv \texttt{TRUE}$. \\

\textbf{Part (d):} $\lnot(M \lor L) \land L \equiv (\lnot M \land \lnot L) \land L \equiv \mathord{\underbrace{(\lnot L \land L)}_{\text{contradiction}}} \land L \equiv \texttt{FALSE}$. \\

\textbf{Part (e):} $(P \rightarrow Q) \lor Q \equiv (\lnot P \lor Q) \lor Q \equiv (Q \lor Q) \lor \lnot P \equiv Q \lor \lnot P$ \\

\textbf{Part (f):} $\lnot (X \rightarrow Y) \lor Y \equiv (X \land \lnot Y) \lor Y \equiv \mathord{\underbrace{(Y \lor \lnot Y)}_{\text{tautology}}} \land (Y \lor X) \equiv X \lor Y$



\end{homeworkProblem}

\begin{homeworkProblem}
Suppose that the possible values of $x$ are all people. Let $Y (x) = $ “x has green hair,” let $Z(x)
= $ “x likes pickles” and $W (x) = $ “$x$ has a pet frog.” Translate the following statements into
words.

\begin{enumerate}[label=(\alph*), topsep=5pt, itemsep=5pt]
    \item $(\forall x) Y(x)$.
    \item $(\exists x) Z(x)$.
    \item $(\forall x)[W(x) \wedge Z(x)]$.
    \item $(\exists x)[Y(x) \rightarrow W(x)]$.
    \item $(\forall x)[W(x) \leftrightarrow \neg Z(x)]$.
\end{enumerate} \vspace{1\baselineskip}

\solution

\textbf{Part (a):} Every person has green hair. \\

\textbf{Part (b):} At least one person likes pickles. // Some people like pickles. \\

\textbf{Part (c):} All people have a pet frog and like pickles. \\
 
\textbf{Part (d):} For some people, if they have green hair, then they have a pet frog.  \\

\textbf{Part (e):} Every person has a pet frog if and only if they do not like pickles.

\end{homeworkProblem}
