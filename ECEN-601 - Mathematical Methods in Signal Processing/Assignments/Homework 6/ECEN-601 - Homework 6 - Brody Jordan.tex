\begin{homeworkProblem}

    Consider \R[4] with the standard inner product. Let W be the subspace of \R[4] consisting of all
    vectors which are orthogonal to both $\alpha = (1, 0, -1, 1)$ and $\beta = (2, 3, -1, 2)$. Find a basis for
    $W$. \\

    \solution

    Constructing a matrix from these vectors:
    \[
        \begin{bmatrix}
            1 & 0 & -1 & 1 \\
            2 & 3 & -1 & 2 \\
        \end{bmatrix} \quad\to\quad
        \begin{bmatrix}
            1 & 0 & -1 & 1 \\
            0 & 3 & 1  & 0 \\
        \end{bmatrix} \quad\to\quad
        \begin{bmatrix}
            1 & 0 & -1           & 1 \\
            0 & 1 & \sfrac{1}{3} & 0 \\
        \end{bmatrix}
    \]

    From these pivot values, $x_1 = x_3 - x_4$ and $3x_2 = -x_3$.
    \[
        \begin{bmatrix}
            x_3 - x_4 \\
            -x_3/3    \\
            x_3       \\
            x_4
        \end{bmatrix} \quad\to\quad
        x_3
        \begin{bmatrix}
            1             \\
            -\sfrac{1}{3} \\
            1             \\
            0             \\
        \end{bmatrix} +
        x_4
        \begin{bmatrix}
            -1 \\
            0  \\
            0  \\
            1  \\
        \end{bmatrix}
    \]
    Thus, a basis for $W$ is
    \[
        \left\{
        \begin{bmatrix}
            1             \\
            -\sfrac{1}{3} \\
            1             \\
            0             \\
        \end{bmatrix},
        \begin{bmatrix}
            -1 \\
            0  \\
            0  \\
            1  \\
        \end{bmatrix}
        \right\}
    \]

\end{homeworkProblem}

% ------ PROBLEM 2 ------ %
\begin{homeworkProblem}

    Let $V$ be an inner product space, and let $\alpha$ and $\beta$ be vectors in $V$. Show that $\alpha = \beta$ if and
    only if $\langle\alpha \mid \gamma\rangle = \langle\beta \mid \gamma\rangle$ for every $\gamma$ in $V$. \\

    \solution

    By the definition of inner product, we know $\langle u + v, w \rangle = \langle u, w \rangle + \langle v, w \rangle$. Then, if we have $\alpha = \beta$, then
    \begin{align*}
        \langle \alpha \mid \gamma \rangle = \langle \beta \mid \gamma \rangle \implies \langle \alpha \mid \gamma \rangle - \langle \beta \mid \gamma \rangle & = 0 \\
        \langle \alpha - \beta \mid \gamma \rangle                                                                                                             & = 0
    \end{align*}
    Inner product orthogonality requires $\gamma = \alpha - \beta$, and since $0$-equivalence means two vectors are equivalent, $\alpha = \beta$.

\end{homeworkProblem}

% ------ PROBLEM 3 ------ %
\begin{homeworkProblem}

    Let
    \begin{equation*}
        \mathbf{p}_1 =
        \begin{bmatrix}
            1 \\
            2 \\
            3 \\
            4 \\
        \end{bmatrix} \quad
        \mathbf{p}_2 =
        \begin{bmatrix}
            4  \\
            -2 \\
            -6 \\
            -7 \\
        \end{bmatrix} \quad
        \mathbf{p}_3 =
        \begin{bmatrix}
            3  \\
            4  \\
            -2 \\
            1  \\
        \end{bmatrix}
    \end{equation*}
    and
    \[
        \mathbf{x} = \begin{bmatrix}
            1 \\
            2 \\
            3 \\
            7 \\
        \end{bmatrix}.
    \]
    Determine the best approximation, $\hat{\mathbf{x}}$, of the vector $\mathbf{x}$ by vectors in $W = \spn\{\mathbf{p}_1, \mathbf{p}_2, \mathbf{p}_3\}$.
    Also, determine the projection of $\mathbf{x}$ onto the orthogonal complement of $W$. \\

    \solution

    We have matrix $A$
    \[
        \begin{bmatrix}
            1 & 4  & 3  \\
            2 & -2 & 4  \\
            3 & -6 & -2 \\
            4 & -7 & 1  \\
        \end{bmatrix}
    \]

    Finding the Gramian matrix, $G$ (also $A^H A$):
    \[
        \begin{bmatrix}
            \inner{\mathbf{p}_1}{\mathbf{p}_1} & \inner{\mathbf{p}_1}{\mathbf{p}_2} & \inner{\mathbf{p}_1}{\mathbf{p}_3} \\
            \inner{\mathbf{p}_2}{\mathbf{p}_1} & \inner{\mathbf{p}_2}{\mathbf{p}_2} & \inner{\mathbf{p}_2}{\mathbf{p}_3} \\
            \inner{\mathbf{p}_3}{\mathbf{p}_1} & \inner{\mathbf{p}_3}{\mathbf{p}_2} & \inner{\mathbf{p}_3}{\mathbf{p}_3}
        \end{bmatrix} =
        \begin{bmatrix}
            30  & -46 & 9  \\
            -46 & 105 & 9  \\
            9   & 9   & 30 \\
        \end{bmatrix}
    \]
    Then the best approximation is $\hat{\mathbf{x}} = AG^{-1}A^T \mathbf{x}$.
    \[
        \hat{\mathbf{x}} =
        \begin{bmatrix}
            1 & 4  & 3  \\
            2 & -2 & 4  \\
            3 & -6 & -2 \\
            4 & -7 & 1  \\
        \end{bmatrix}
        \begin{bmatrix}
            30  & -46 & 9  \\
            -46 & 105 & 9  \\
            9   & 9   & 30 \\
        \end{bmatrix}^{-1}
        \begin{bmatrix}
            1 & 4  & 3  \\
            2 & -2 & 4  \\
            3 & -6 & -2 \\
            4 & -7 & 1  \\
        \end{bmatrix}^T
        \begin{bmatrix}
            1 \\
            2 \\
            3 \\
            7 \\
        \end{bmatrix}
    \]
    By WolframAlpha, $\hat{\mathbf{x}} \approx (0.89, 2.97, 4.12, 5.70)^T$.


\end{homeworkProblem}

% ------ PROBLEM 4 ------ %
\begin{homeworkProblem}

    Let $S$ be a subset of an inner product space $V$. Show that $(S^\perp)^\perp$ contains the subspace
    spanned by $S$. When $V$ is finite-dimensional, show that $(S^\perp)^\perp$ is the subspace spanned by $S$. \\

    \solution

    We know that $S^\perp = \{\vec{v} \in V \mid \langle \vec{v} \mid \vec{s} \rangle = 0, \forall \vec{s} \in S\}$. Then $(S^\perp)^\perp = \{\vec{v} \in V \mid \langle \vec{v} \mid \vec{w} \rangle = 0, \forall \vec{w} \in S^\perp\}$.
    By these definitions, every vector in $S$ is orthogonal to every vector in $S^\perp$. \\

    This is equivalent to saying $s \in (S^\perp)^\perp, \forall s \in S$. Thus, the span of $S$ is contained in $(S^\perp)^\perp$. \\

\end{homeworkProblem}

% ------ PROBLEM 5 ------ %
\begin{homeworkProblem}

    Let $V$ be a finite-dimensional inner product space, and let $\mathcal{B} = (\alpha_1, \ldots, \alpha_n)$ be an orthonormal
    basis for $V$. Let $T$ be a linear operator on $V$; and $A$, the matrix of $T$ in the ordered basis $\mathcal{B}$.
    Prove that
    \[
        A_{ij} = \langle T \alpha_j \mid \alpha_i \rangle
    \]

    \solution

    With orthonormal basis $\mathcal{B} = (\alpha_1, \ldots, \alpha_n)$, we have
    \[
        \vec{v} = \sum_{k=1}^n \langle \vec{v} \mid \alpha_k \rangle \alpha_k
    \]
    Applying the transformation shows us that
    \[
        T \alpha_j = \sum_{k=1}^n A_{kj} \alpha_k
    \]
    but since
    \[
        T \alpha_j = \sum_{k=1}^n \langle T \alpha_j \mid \alpha_k \rangle \alpha_k
    \]
    we have that
    \[
        A_{ij} = \langle T \alpha_j \mid \alpha_i \rangle
    \]

\end{homeworkProblem}

% ------ PROBLEM 6 ------ %
\begin{homeworkProblem}

    Suppose $V = W_1 \oplus W_2$ and that $f_1$ and $f_2$ are inner products on $W_1$ and $W_2$, respectively.
    Show that there is a unique inner product $f$ on $V$ such that
    \begin{enumerate}[label=(\alph*), topsep=5pt, itemsep=5pt]
        \item $W_2 = W_1^\perp$;
        \item $f(\alpha, \beta) = f_k(\alpha, \beta)$, when $\alpha$, $\beta$ are in $W_k, k = 1, 2$. \newline
    \end{enumerate}

    \solution

    With $V = W_1 \oplus W_2$, we have $\forall v \in V$, $v = w_1 + w_2$ with $w_1 \in W_1$ and $w_2 \in W_2$. \\

    We have $f(\alpha_1, \beta_1) = f_1(\alpha_1, \beta_1)$ for $\alpha_1, \beta_1 \in W_1$ and $f(\alpha_2, \beta_2) = f_2(\alpha_2, \beta_2)$ for $\alpha_2, \beta_2 \in W_2$. \\

    By linearity of inner products, we have
    \begin{align*}
        f(\alpha_1 + \alpha_2, \beta_1 + \beta_2) & = f(\alpha_1, \beta_1) + f(\alpha_1, \beta_2) + f(\alpha_2, \beta_1) + f(\alpha_2, \beta_2) \\
                                                  & = f_1(\alpha_1, \beta_1) + 0 + 0 + f_2(\alpha_2, \beta_2)                                   \\
                                                  & = f_1(\alpha_1, \beta_1) + f_2(\alpha_2, \beta_2)
    \end{align*}

\end{homeworkProblem}
