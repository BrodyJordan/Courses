\begin{homeworkProblem}
    Consider the problem
    \begin{align*}
        \text{minimize:} \quad   & x_1^2 + x_2^2                    \\
        \text{subject to:} \quad & (x_1 - 1)^2 + (x_2 - 2)^2 \leq 2 \\
                                 & 2x_1 + x_2 \geq 3                \\
                                 & x_1 \geq 0, \, x_2 \geq 0.
    \end{align*}
    \begin{enumerate}[label=(\alph*), topsep=5pt, itemsep=5pt]
        \item Derive the KKT conditions. Find a solution for these conditions. Is this the optimal solution?
        \item Does the strong duality hold?
        \item Compare the solution to the one you obtained last time using geometry. Are they the same?
        \item Consider the problem
              \begin{align*}
                  \text{minimize:} \quad   & x_1^2 + x_2^2                            \\
                  \text{subject to:} \quad & (x_1 - 1)^2 + (x_2 - 2)^2 \leq 2 + \mu_1 \\
                                           & 2x_1 + x_2 \geq 3 + \mu_2                \\
                                           & x_1 \geq 0, \, x_2 \geq 0.
              \end{align*}
              The optimal value is a function of $\mu_1$ and $\mu_2$, i.e., $p^*(\mu_1, \mu_2)$. Compute
              \[
                  \frac{\partial p^*(\mu_1, \mu_2)}{\partial \mu_1} \bigg{|}_{(0, 0)}
              \]
              \[
                  \frac{\partial p^*(\mu_1, \mu_2)}{\partial \mu_2} \bigg{|}_{(0, 0)}
              \] \newline
    \end{enumerate}

    \solution

    \textbf{Part (a):} The Lagrangian:
    \[
        L(x, \lambda) = x_1^2 + x_2^2 + \lambda_1 \left( (x_1 - 1)^2 + (x_2 - 2)^2 - 2 \right) + \lambda_2 (3 - 2x_1 - x_2) + \lambda_3 (-x_1) + \lambda_4 (-x_2)
    \]
    and our conditions:
    \[
        \begin{cases}
            \lambda_1 \left( (x_1 - 1)^2 + (x_2 - 2)^2 - 2 \right) = 0 \\
            \lambda_2 (3 - 2x_1 - x_2) = 0                             \\
            \lambda_3 (-x_1) = 0                                       \\
            \lambda_4 (-x_2) = 0                                       \\
            2x_1 + 2 \lambda_1(x_1 - 1) - 2 \lambda_2 - \lambda_3 = 0  \\
            2x_2 + 2 \lambda_1(x_2 - 2) - \lambda_2 - \lambda_4 = 0    \\
            \lambda \succeq 0                                          \\
        \end{cases}
    \]

    \pagebreak

    \textbf{Part (b):} The strong duality holds since Slater's condition is satisfied. \\

    \textbf{Part (c):} This form has the same solution as the solution obtained by geometric analysis $x^{\star} = (1.2, 0.6)^T$. \\

    \textbf{Part (d):} By local sensitivity, we have
    \[
        \lambda_i^* = - \frac{\partial p^*(\mu_1, \mu_2)}{\partial \mu_i} \bigg{|}_{(0, 0)}
    \]
    Therefore
    \[
        \frac{\partial p^*(\mu_1, \mu_2)}{\partial \mu_1} \bigg{|}_{(0, 0)} = -\lambda_1^* = 0 \quad \text{and} \quad \frac{\partial p^*(\mu_1, \mu_2)}{\partial \mu_2} \bigg{|}_{(0, 0)} = -\lambda_2^*
    \]


\end{homeworkProblem}

\begin{homeworkProblem}

    Consider the following optimization problem:
    \begin{align*}
        \text{minimize:} \quad   & \sum_{i=1}^n \alpha_{i} \max \left\{ \ln \frac{\sigma_i^2}{x_i}, 0 \right\} \\
        \text{subject to:} \quad & \sum_{i=1}^n \beta_{i} x_{i} \leq D,
    \end{align*}
    where $x = (x_1, x_2, \ldots, x_n) \succ 0$ are the decision variables. Assume $\sigma^2 \succ 0$, $\alpha \succeq 0$, and $\beta \succ 0$.

    \begin{enumerate}[label=(\alph*), topsep=5pt, itemsep=5pt]
        \item Argue that this problem is convex;
        \item Write down the KKT conditions;
        \item Provide an explicit solution for this problem (in terms of certain intermediate variable), by providing a solution to the KKT conditions;
        \item Show that equivalently, the inequality constraint can be replaced by an equality constraint for any $0 < D \leq \sum_{i=1}^n \beta_{i} \sigma_{i}^2$;
        \item Let $n=3$, $(\sigma_1^2, \sigma_2^2, \sigma_3^2) = (1, 4, 25)$, $\alpha = (1,1,2)$, and $\beta = (2, 1, 1).$ Consider the optimal value $p^*$ as a function of the parameter $D$. Find this function in a closed form.
        \item Design an algorithm to compute the optimal solution for any given $\sigma^2 \succ 0$, $\alpha \succeq 0$, and $\beta \succ 0$, with a computation complexity at most $O(n)$. \newline
    \end{enumerate}

    \solution

    \textbf{Part (a):} This problem is convex as we can rewrite the $\max$ in the objective function as a constraint instead, which results in $\ln \frac{\sigma_i^2}{x_i}$ with the condition that $x_i \leq \sigma_i^2$. \\

    \textbf{Part (b):} The Lagrangian:
    \[
        L(x, \lambda ) = \sum_{i=1}^n \alpha_{i} \ln \frac{\sigma_i^2}{x_i} + \lambda_1 \left( \sum_{i=1}^n \beta_{i} x_{i} - D \right) + \lambda_2 (x_i - \sigma_i^2)
    \]
    and our KKT conditions:
    \[
        \begin{cases}
            \lambda_1 \left( \sum_{i=1}^n \beta_{i} x_{i} - D \right) = 0 \\
            \lambda_2 (x_i - \sigma_i^2) = 0                              \\
            \lambda \succeq 0                                             \\
            \frac{-\alpha}{x_i} + \lambda_1 \beta_i + \lambda_2 x_i = 0
        \end{cases}
    \]
    \textbf{Part (c):} $x_i^* = -\alpha_i/\beta_i$. \\

    \textbf{Part (d):} Since we impose the condition $x_i \leq \sigma_i^2$, then we can rewrite the original constraint condition as $0 < D \leq \sum_{i=1}^n \beta_{i} \sigma_{i}^2$. \\

    \pagebreak

    \textbf{Part (e):} $n = 3$, $(\sigma_1^2, \sigma_2^2, \sigma_3^2) = (1, 4, 25)$, $\alpha = (1,1,2)$, and $\beta = (2, 1, 1).$
    \begin{align*}
        f_0(x) & = \max \left\{ \ln \frac{1}{x_1}, 0 \right\} + \max \left\{ \ln \frac{4}{x_2}, 0 \right\} + 2 \max \left\{ \ln \frac{25}{x_3}, 0 \right\} \\
        f_1(x) & = 2x_1 + x_2 + x_3 \leq D
    \end{align*}

    \textbf{Part (f):} We want the largest $x_i$ values that lie on the plane $2x_1 + x_2 + x_3 = D$.

\end{homeworkProblem}

\begin{homeworkProblem}

    Strong duality holds for the problem
    \begin{align*}
        \text{minimize:} \quad   & -4x_1^2 + x_2^2 + x_3^2 + 2(x_1 + x_2 + x_3) \\
        \text{subject to:} \quad & x_1^2 - 2x_2^2 + x_3^2 \leq 1,
    \end{align*}

    although the problem is not convex (see Appendix B.1). Derive the KKT conditions. Find all
    solutions for the KKT conditions. Which one is the optimal value (assuming primal and dual
    optimal values both attained)? You may need to use a program (Matlab, Mathematica, or
    Python package) to plot and solve a non-linear equation; approximate solution is acceptable. \\

    \solution

    The Lagrangian:
    \[
        L(x, \lambda) = -4x_1^2 + x_2^2 + x_3^2 + 2(x_1 + x_2 + x_3) + \lambda (x_1^2 - 2x_2^2 + x_3^2 - 1)
    \]
    KKT conditions:
    \begin{enumerate}[label=(\alph*), topsep=5pt, itemsep=5pt]
        \item Primal constraints: $x_1^2 - 2x_2^2 + x_3^2 - 1 \leq 0$
        \item Dual constraints: $\lambda \succeq 0$
        \item Complementary slackness: $\lambda (x_1^2 - 2x_2^2 + x_3^2 - 1) = 0$
        \item Vanishing gradient:
              \begin{equation*}
                  \nabla f_0(x) + \lambda \nabla f_1(x) = 0 ~ ~ \implies ~ ~
                  \begin{cases}
                      -8x_1 + 2 \lambda x_1 = 0   \\
                      2x_2 + 2 - \lambda 4x_2 = 0 \\
                      2x_3 + 2 + \lambda 2x_3 = 0 \\
                  \end{cases}
              \end{equation*}
    \end{enumerate}
    Solving these equations, we get
    \[
        x_1 = \frac{1}{4 - \lambda}, \quad x_2 = \frac{1}{2\lambda-1}, \quad x_3 = \frac{-1}{\lambda + 1}.
    \]
    Plugging these into the complementary slackness condition, we get
    \[
        \lambda \left( \frac{1}{(4 - \lambda)^2} - 2 \left( \frac{1}{2\lambda-1} \right)^2 + \left( \frac{-1}{\lambda + 1} \right)^2 - 1 \right) = 0.
    \]
    The solution (evaluated using WolframAlpha) is $\lambda \approx 3.0085$ and $\lambda \approx 5.00155$, which gives
    \[
        x^{\star} \approx (1.00857, 0.199322, -0.24947)^T \quad \text{and} \quad x^{\star} \approx (-0.998452, 0.111073, -0.166624)^T.
    \]
\end{homeworkProblem}

\begin{homeworkProblem}

    Consider the following problem
    \begin{align*}
        \text{minimize:} \quad   & \sum_{i=1}^k \phi(|r|_{[i]}) \\
        \text{subject to:} \quad & r = Ax - b,
    \end{align*}
    where $|r|_{[1]} \geq |r|_{[2]} \geq \ldots \geq |r|_{[m]}$ are the numbers $|r_1|, |r_2|, \ldots, |r_m|$ sorted in decreasing order. Here matrix $A$ is $m\times n$.
    \begin{enumerate}[label=(\alph*), topsep=5pt, itemsep=5pt]
        \item When $\phi(u)=u$, reformulate the problem into an LP. The problem with this penalty function minimizes the sum of the largest $k$ residuals. (\textit{Hint}: you need the formulation in exercise 5.19.)
        \item When $\phi(u)$ is the deadzone-linear penalty function
              \[
                  \phi(u) = \begin{dcases}
                      0,       & |u| \leq a; \\
                      |u| - a, & |u| > a;    \\
                  \end{dcases}
              \]
              where $a > 0$, reformulate the problem into an LP. This penalty function minimizes the
              sum of the largest $k$ residuals, but with a deadzone. Recall we have discussed a problem
              similar to this in class. \newline
    \end{enumerate}

    \solution

    \textbf{Part (a):} As an LP,
    \begin{align*}
        \text{minimize:} \quad   & \mathbf{1}^T r \\
        \text{subject to:} \quad & r = Ax - b     \\
                                 & r \succeq 0    \\
    \end{align*}

    \textbf{Part (b):}
    \begin{align*}
        \text{minimize:} \quad   & \sum_{i=1}^{k} t_i       \\
        \text{subject to:} \quad & -y \leq |r|_{[i]} \leq y \\
                                 & -t \leq y - |a| \leq t
    \end{align*}

\end{homeworkProblem}
